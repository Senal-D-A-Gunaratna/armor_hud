[Mineclonia Armor HUD]

# Horizontal anchor for the armor HUD. The group of equipped armor pieces is
# laid out from this edge or from the screen centre.
# left:   start at the left edge, growing right
# center: centre the group on the screen
# right:  start at the right edge, growing left
mcl_armor_hud_anchor (Horizontal anchor) enum left left,center,right

# Scale of the 16x16 armor item icons. 1.0 draws each icon at 16x16 pixels.
mcl_armor_hud_icon_scale (Icon scale) float 1.0 0.25 4.0

# Interval in milliseconds between visibility changes of the critical icon
# blink. A value of 500 makes the icon switch between visible and hidden
# every 500 ms (a full on/off cycle takes 1000 ms). The durability bar stays
# visible at all times.
mcl_armor_hud_blink_interval (Blink interval, ms) int 500 100 10000

[Armor HUD]

# Horizontal anchor for the armor HUD. Armor pieces are always laid out on the
# left, beside the mcl_offhand slot, and shift left when an offhand item makes
# the offhand HUD appear.
armor_hud_anchor (Horizontal anchor) enum left left

# Interval in milliseconds between visibility changes of the critical icon
# blink. A value of 200 makes the icon switch between visible and hidden
# every 200 ms (a full on/off cycle takes 400 ms). The durability bar stays
# visible at all times.
armor_hud_blink_interval (Blink interval, ms) int 200 100 10000

# Durability percentage at or below which an armor piece's icon blinks while
# its durability bar stays visible. 0 effectively disables blinking.
armor_hud_critical_threshold (Critical durability threshold, %) int 10 0 100
